Example algorithms

BFS & DFS traversal

void bfs(int src, vector<vector<int>>& adj, vector<bool>& visited) {
    queue<int> q;
    q.push(src);
    visited[src] = true;
    while (!q.empty()) {
        int node = q.front(); q.pop();
        for (int next : adj[node]) {
            if (!visited[next]) { visited[next] = true; q.push(next); }
        }
    }
}

void dfs(int node, vector<vector<int>>& adj, vector<bool>& visited) {
    visited[node] = true;
    for (int next : adj[node])
        if (!visited[next]) dfs(next, adj, visited);
}

Dijkstra's Algorithm

vector<int> dijkstra(int src, int n, vector<vector<pair<int,int>>>& adj) {
    vector<int> dist(n, INT_MAX);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> pq;
    dist[src] = 0;
    pq.push({0, src});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;
        for (auto& [v, w] : adj[u]) {
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }
    return dist;
}

Bellman–Ford — handles negative weights

Relax every edge V−1 times; if anything still relaxes on the Vth pass, there's a negative cycle.

struct Edge { int u, v, weight; };

vector<long long> bellmanFord(int n, int src, vector<Edge>& edges) {
    vector<long long> dist(n, LLONG_MAX);
    dist[src] = 0;

    for (int i = 0; i < n - 1; i++) {
        for (auto& e : edges) {
            if (dist[e.u] != LLONG_MAX && dist[e.u] + e.weight < dist[e.v]) {
                dist[e.v] = dist[e.u] + e.weight;
            }
        }
    }

    // one extra pass: if a distance still improves, there's a negative cycle
    for (auto& e : edges) {
        if (dist[e.u] != LLONG_MAX && dist[e.u] + e.weight < dist[e.v]) {
            throw runtime_error("negative weight cycle detected");
        }
    }
    return dist;
}

Topological Sort — Kahn's algorithm (BFS)

vector<int> topoSort(int n, vector<vector<int>>& adj) {
    vector<int> indegree(n, 0);
    for (int u = 0; u < n; u++)
        for (int v : adj[u]) indegree[v]++;

    queue<int> q;
    for (int i = 0; i < n; i++) if (indegree[i] == 0) q.push(i);

    vector<int> order;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        order.push_back(u);
        for (int v : adj[u])
            if (--indegree[v] == 0) q.push(v);
    }
    return order;
}

Union–Find (Disjoint Set) with path compression

struct DSU {
    vector<int> parent, rank_;
    DSU(int n) : parent(n), rank_(n, 0) {
        iota(parent.begin(), parent.end(), 0);
    }
    int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }
    void unite(int x, int y) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return;
        if (rank_[rx] < rank_[ry]) swap(rx, ry);
        parent[ry] = rx;
        if (rank_[rx] == rank_[ry]) rank_[rx]++;
    }
};

Number of Islands (flood fill)

void floodFill(vector<vector<char>>& g, int i, int j) {
    if (i < 0 || j < 0 || i >= (int)g.size() || j >= (int)g[0].size() || g[i][j] != '1') return;
    g[i][j] = '0';
    floodFill(g, i + 1, j); floodFill(g, i - 1, j); floodFill(g, i, j + 1); floodFill(g, i, j - 1);
}
int numIslands(vector<vector<char>>& g) {
    int count = 0;
    for (int i = 0; i < (int)g.size(); i++)
        for (int j = 0; j < (int)g[0].size(); j++)
            if (g[i][j] == '1') { count++; floodFill(g, i, j); }
    return count;
}

Shortest Path in an Unweighted Grid (BFS)

int shortestPathGrid(vector<vector<int>>& g, int sr, int sc, int tr, int tc) {   // 0 free, 1 wall
    int R = g.size(), C = g[0].size();
    if (g[sr][sc] || g[tr][tc]) return -1;
    vector<vector<int>> dist(R, vector<int>(C, -1));
    queue<pair<int,int>> q;
    dist[sr][sc] = 0; q.push({sr, sc});
    int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
    while (!q.empty()) {
        auto [r, c] = q.front(); q.pop();
        if (r == tr && c == tc) return dist[r][c];
        for (int d = 0; d < 4; d++) {
            int nr = r + dr[d], nc = c + dc[d];
            if (nr < 0 || nc < 0 || nr >= R || nc >= C || g[nr][nc] || dist[nr][nc] != -1) continue;
            dist[nr][nc] = dist[r][c] + 1;
            q.push({nr, nc});
        }
    }
    return -1;
}

Rotten Oranges (multi-source BFS)

int orangesRotting(vector<vector<int>>& g) {
    int R = g.size(), C = g[0].size(), fresh = 0, minutes = 0;
    queue<pair<int,int>> q;
    for (int i = 0; i < R; i++)
        for (int j = 0; j < C; j++) {
            if (g[i][j] == 2) q.push({i, j});
            else if (g[i][j] == 1) fresh++;
        }
    int dr[] = {1, -1, 0, 0}, dc[] = {0, 0, 1, -1};
    while (!q.empty() && fresh) {
        int sz = q.size();
        while (sz--) {
            auto [r, c] = q.front(); q.pop();
            for (int d = 0; d < 4; d++) {
                int nr = r + dr[d], nc = c + dc[d];
                if (nr < 0 || nc < 0 || nr >= R || nc >= C || g[nr][nc] != 1) continue;
                g[nr][nc] = 2; fresh--; q.push({nr, nc});
            }
        }
        minutes++;
    }
    return fresh ? -1 : minutes;
}

Cycle Detection: Undirected (parent tracking) and Directed (recursion stack)

bool hasCycleUndirected(int u, int parent, vector<vector<int>>& adj, vector<bool>& vis) {
    vis[u] = true;
    for (int v : adj[u]) {
        if (!vis[v]) { if (hasCycleUndirected(v, u, adj, vis)) return true; }
        else if (v != parent) return true;
    }
    return false;
}
bool hasCycleDirected(int u, vector<vector<int>>& adj, vector<int>& state) {   // 0 new, 1 on stack, 2 done
    state[u] = 1;
    for (int v : adj[u]) {
        if (state[v] == 1) return true;
        if (state[v] == 0 && hasCycleDirected(v, adj, state)) return true;
    }
    state[u] = 2;
    return false;
}

Bipartite Check (BFS 2-coloring)

bool isBipartite(vector<vector<int>>& adj) {
    int n = adj.size();
    vector<int> color(n, -1);
    for (int s = 0; s < n; s++) {
        if (color[s] != -1) continue;
        queue<int> q; q.push(s); color[s] = 0;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (int v : adj[u]) {
                if (color[v] == -1) { color[v] = color[u] ^ 1; q.push(v); }
                else if (color[v] == color[u]) return false;
            }
        }
    }
    return true;
}

Prim's MST

long long primMST(int n, vector<vector<pair<int,int>>>& adj) {      // adj[u] = {v, w}
    vector<bool> inMST(n, false);
    priority_queue<pair<int,int>, vector<pair<int,int>>, greater<>> pq;
    pq.push({0, 0});
    long long total = 0;
    while (!pq.empty()) {
        auto [w, u] = pq.top(); pq.pop();
        if (inMST[u]) continue;
        inMST[u] = true; total += w;
        for (auto& [v, wt] : adj[u]) if (!inMST[v]) pq.push({wt, v});
    }
    return total;
}

Floyd–Warshall (all-pairs)

const long long INF = 1e18;
void floydWarshall(vector<vector<long long>>& d) {      // d[i][i] = 0, no edge = INF
    int n = d.size();
    for (int k = 0; k < n; k++)
        for (int i = 0; i < n; i++)
            for (int j = 0; j < n; j++)
                if (d[i][k] < INF && d[k][j] < INF) d[i][j] = min(d[i][j], d[i][k] + d[k][j]);
}

Clone a Graph

struct GNode { int val; vector<GNode*> neighbors; GNode(int v) : val(v) {} };

GNode* cloneGraph(GNode* node) {
    if (!node) return nullptr;
    unordered_map<GNode*, GNode*> copy;
    queue<GNode*> q;
    copy[node] = new GNode(node->val); q.push(node);
    while (!q.empty()) {
        GNode* cur = q.front(); q.pop();
        for (GNode* nb : cur->neighbors) {
            if (!copy.count(nb)) { copy[nb] = new GNode(nb->val); q.push(nb); }
            copy[cur]->neighbors.push_back(copy[nb]);
        }
    }
    return copy[node];
}

Word Ladder

int ladderLength(string begin, string end, vector<string>& wordList) {
    unordered_set<string> dict(wordList.begin(), wordList.end());
    if (!dict.count(end)) return 0;
    queue<pair<string,int>> q;
    q.push({begin, 1}); dict.erase(begin);
    while (!q.empty()) {
        auto [word, steps] = q.front(); q.pop();
        if (word == end) return steps;
        for (int i = 0; i < (int)word.size(); i++) {
            char orig = word[i];
            for (char c = 'a'; c <= 'z'; c++) {
                word[i] = c;
                if (dict.count(word)) { dict.erase(word); q.push({word, steps + 1}); }
            }
            word[i] = orig;
        }
    }
    return 0;
}

Alien Dictionary (graph from adjacent words + Kahn)

string alienOrder(vector<string>& words) {
    vector<vector<int>> adj(26);
    vector<int> indeg(26, 0); vector<bool> present(26, false);
    for (auto& w : words) for (char c : w) present[c - 'a'] = true;
    for (int i = 0; i + 1 < (int)words.size(); i++) {
        string &a = words[i], &b = words[i + 1];
        int len = min(a.size(), b.size()), j = 0;
        while (j < len && a[j] == b[j]) j++;
        if (j == len) { if (a.size() > b.size()) return ""; continue; }   // "abc" before "ab" is invalid
        adj[a[j] - 'a'].push_back(b[j] - 'a');
        indeg[b[j] - 'a']++;
    }
    queue<int> q; string res; int total = 0;
    for (int i = 0; i < 26; i++) if (present[i]) { total++; if (!indeg[i]) q.push(i); }
    while (!q.empty()) {
        int u = q.front(); q.pop();
        res += char('a' + u);
        for (int v : adj[u]) if (--indeg[v] == 0) q.push(v);
    }
    return (int)res.size() == total ? res : "";
}

Critical Connections, Tarjan's bridges

int timer_ = 0;
void bridgeDfs(int u, int parent, vector<vector<int>>& adj, vector<int>& tin, vector<int>& low,
               vector<vector<int>>& bridges) {
    tin[u] = low[u] = ++timer_;
    for (int v : adj[u]) {
        if (v == parent) continue;
        if (tin[v]) low[u] = min(low[u], tin[v]);
        else {
            bridgeDfs(v, u, adj, tin, low, bridges);
            low[u] = min(low[u], low[v]);
            if (low[v] > tin[u]) bridges.push_back({u, v});
        }
    }
}
vector<vector<int>> criticalConnections(int n, vector<vector<int>>& edges) {
    vector<vector<int>> adj(n), bridges;
    for (auto& e : edges) { adj[e[0]].push_back(e[1]); adj[e[1]].push_back(e[0]); }
    vector<int> tin(n, 0), low(n, 0);
    timer_ = 0;
    for (int i = 0; i < n; i++) if (!tin[i]) bridgeDfs(i, -1, adj, tin, low, bridges);
    return bridges;
}

Cheapest Flights Within K Stops (Bellman–Ford, K+1 rounds)

int findCheapestPrice(int n, vector<vector<int>>& flights, int src, int dst, int k) {
    vector<int> dist(n, INT_MAX);
    dist[src] = 0;
    for (int i = 0; i <= k; i++) {
        vector<int> tmp = dist;
        for (auto& f : flights) {
            int u = f[0], v = f[1], w = f[2];
            if (dist[u] != INT_MAX && dist[u] + w < tmp[v]) tmp[v] = dist[u] + w;
        }
        dist = tmp;
    }
    return dist[dst] == INT_MAX ? -1 : dist[dst];
}
