Example algorithms

Trie — insert, search, startsWith

struct TrieNode {
    TrieNode* children[26] = {};
    bool isEnd = false;
};

class Trie {
    TrieNode* root = new TrieNode();
public:
    void insert(const string& word) {
        TrieNode* node = root;
        for (char c : word) {
            int i = c - 'a';
            if (!node->children[i]) node->children[i] = new TrieNode();
            node = node->children[i];
        }
        node->isEnd = true;
    }

    bool search(const string& word) {
        TrieNode* node = find(word);
        return node && node->isEnd;
    }

    bool startsWith(const string& prefix) {
        return find(prefix) != nullptr;
    }

private:
    TrieNode* find(const string& s) {
        TrieNode* node = root;
        for (char c : s) {
            int i = c - 'a';
            if (!node->children[i]) return nullptr;
            node = node->children[i];
        }
        return node;
    }
};

Maximum XOR of Two Numbers (binary trie)

struct BitTrie {
    struct Node { Node* ch[2] = {nullptr, nullptr}; };
    Node* root = new Node();
    void insert(int x) {
        Node* cur = root;
        for (int b = 30; b >= 0; b--) {
            int bit = (x >> b) & 1;
            if (!cur->ch[bit]) cur->ch[bit] = new Node();
            cur = cur->ch[bit];
        }
    }
    int maxXorWith(int x) {
        Node* cur = root; int res = 0;
        for (int b = 30; b >= 0; b--) {
            int bit = (x >> b) & 1;
            if (cur->ch[bit ^ 1]) { res |= (1 << b); cur = cur->ch[bit ^ 1]; }
            else cur = cur->ch[bit];
        }
        return res;
    }
};
int findMaximumXOR(vector<int>& a) {
    BitTrie t; int best = 0;
    t.insert(a[0]);
    for (int i = 1; i < (int)a.size(); i++) { best = max(best, t.maxXorWith(a[i])); t.insert(a[i]); }
    return best;
}

Word Search II (trie + grid DFS)

struct WNode { WNode* next[26] = {}; string word; };

void wordDfs2(vector<vector<char>>& b, int i, int j, WNode* node, vector<string>& res) {
    char c = b[i][j];
    if (c == '#' || !node->next[c - 'a']) return;
    node = node->next[c - 'a'];
    if (!node->word.empty()) { res.push_back(node->word); node->word.clear(); }   // avoid duplicates
    b[i][j] = '#';
    if (i > 0) wordDfs2(b, i - 1, j, node, res);
    if (j > 0) wordDfs2(b, i, j - 1, node, res);
    if (i + 1 < (int)b.size())    wordDfs2(b, i + 1, j, node, res);
    if (j + 1 < (int)b[0].size()) wordDfs2(b, i, j + 1, node, res);
    b[i][j] = c;
}
vector<string> findWords(vector<vector<char>>& b, vector<string>& words) {
    WNode* root = new WNode();
    for (auto& w : words) {
        WNode* cur = root;
        for (char c : w) {
            if (!cur->next[c - 'a']) cur->next[c - 'a'] = new WNode();
            cur = cur->next[c - 'a'];
        }
        cur->word = w;
    }
    vector<string> res;
    for (int i = 0; i < (int)b.size(); i++)
        for (int j = 0; j < (int)b[0].size(); j++) wordDfs2(b, i, j, root, res);
    return res;
}

Autocomplete (uses TrieNode from the trie example)

void collect(TrieNode* node, string& cur, vector<string>& out) {
    if (node->isEnd) out.push_back(cur);
    for (int i = 0; i < 26; i++)
        if (node->children[i]) {
            cur.push_back('a' + i);
            collect(node->children[i], cur, out);
            cur.pop_back();
        }
}
vector<string> autocomplete(TrieNode* root, const string& prefix) {
    TrieNode* node = root;
    for (char c : prefix) { node = node->children[c - 'a']; if (!node) return {}; }
    vector<string> out; string cur = prefix;
    collect(node, cur, out);
    return out;
}

Replace Words with Shortest Dictionary Root

string replaceWords(vector<string>& dict, string sentence) {
    TrieNode* root = new TrieNode();
    for (auto& d : dict) {
        TrieNode* cur = root;
        for (char c : d) {
            if (!cur->children[c - 'a']) cur->children[c - 'a'] = new TrieNode();
            cur = cur->children[c - 'a'];
        }
        cur->isEnd = true;
    }
    stringstream ss(sentence); string w, res;
    while (ss >> w) {
        TrieNode* cur = root; string prefix; bool found = false;
        for (char c : w) {
            if (!cur->children[c - 'a']) break;
            cur = cur->children[c - 'a'];
            prefix += c;
            if (cur->isEnd) { found = true; break; }
        }
        if (!res.empty()) res += ' ';
        res += found ? prefix : w;
    }
    return res;
}

Longest Word Built from Other Words (sorted-prefix version, same idea as trie + DFS)

string longestWord(vector<string>& words) {
    sort(words.begin(), words.end());
    unordered_set<string> built = {""};
    string best;
    for (auto& w : words)
        if (built.count(w.substr(0, w.size() - 1))) {
            built.insert(w);
            if (w.size() > best.size()) best = w;
        }
    return best;
}
