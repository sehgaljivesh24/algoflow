Binary Search

Core signal: the search space is sorted or monotonic — even a yes/no feasibility check that flips once as a value increases counts.

1D array — Direct Search

When you see… Pattern
Find an element in a sorted array Standard binary search: lo=0, hi=n-1, while lo≤hi
Find first / last occurrence of X Lower / upper bound: lo<hi, shrink towards X
Count total occurrences Upper bound − lower bound
Search insert position Return lo (first position ≥ target)
Floor and Ceil in sorted array Maintain boundaries while shrinking

1D array — Rotated Sorted Array

When you see… Pattern
Search in rotated sorted array (no duplicates) Check a[lo]≤a[mid], identify sorted half, eliminate unsorted
Search rotated array with duplicates When a[lo]==a[mid]==a[hi], shrink: lo++, hi--
Find minimum in rotated sorted array Compare mid with hi, eliminate larger side
How many times array is rotated Find index of minimum element

1D array — Peak / Extrema / Special Cases

When you see… Pattern
Find a peak element Compare a[mid] with a[mid+1], move toward larger
Single element in sorted array (others ×2) Binary search on index parity (odd/even)
Sqrt(x), Nth root Binary search on answer value: mid² vs x

Binary Search on Answer — Minimize/Maximize

When you see… Pattern
Minimize X (speed, time, cost, capacity) BS on answer + feasibility: hi=mid if valid, else lo=mid+1
Maximize X (distance, hours, allocation) BS on answer + feasibility: lo=mid+1 if valid, else hi=mid
Minimize eating speed (hours ≤ h)(Koko eating bananas) Binary search on speed, check sum of ceil(pile/speed)
Ship packages in D days (min capacity) Binary search on capacity, greedy day packing
Allocate books to K painters (minimize max) Binary search on max pages, greedy allocation
Aggressive cows (maximize min distance) Binary search on distance, greedy placement check
Kth missing positive number Binary search on value, count missing ≤ mid
Minimize max distance to gas station Binary search on distance, count stations needed
Minimize days to make M bouquets Binary search on days, check adjacent bloom feasibility
Find smallest divisor (threshold) Binary search on divisor, sum of divisions check

2D array

When you see… Pattern
Row-wise & column-wise sorted matrix (search) Start top-right corner, move left if big, down if small
Find row with maximum 1's Binary search each row, optimize pointer movement
Matrix sorted like flattened 1D array Treat as 1D: row=mid/cols, col=mid%cols
Kth smallest in sorted matrix Binary search on value, count elements ≤ mid
Peak element in 2D (no larger neighbor) Binary search on column, compare mid neighbors
Matrix median Binary search on value, binary count check

Two Sorted Arrays

When you see… Pattern
Median of two sorted arrays Binary search on partition, balance left/right halves
Kth element of two sorted arrays Binary search on position or elimination approach

Example algorithms

Binary Search — Standard Template

int binarySearch(vector<int>& a, int target) {
    int lo = 0, hi = a.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) return mid;
        else if (a[mid] < target) lo = mid + 1;
        else hi = mid - 1;
    }
    return -1;
}

Lower Bound (First ≥ Target)

int lowerBound(vector<int>& a, int target) {

        if (a[mid] < target) lo = mid + 1;
        else hi = mid;
    }
    return lo;
}

Search in Rotated Sorted Array

int search(vector<int>& a, int target) {
    int lo = 0, hi = a.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) return mid;
        if (a[lo] <= a[mid]) {
            if (a[lo] <= target && target < a[mid]) hi = mid - 1;
            else lo = mid + 1;
        } else {
            if (a[mid] < target && target <= a[hi]) lo = mid + 1;
            else hi = mid - 1;
        }
    }
    return -1;
}

Staircase Walk (2D Sorted Matrix)

bool searchMatrix(vector<vector<int>>& matrix, int target) {
    int row = 0, col = matrix[0].size() - 1;  // top-right
    while (row < matrix.size() && col >= 0) {
        if (matrix[row][col] == target) return true;
        else if (matrix[row][col] > target) col--;
        else row++;
    }
    return false;
}

Upper bound, first/last occurrence, count (uses lowerBound from above)

int upperBound(vector<int>& a, int target) {      // first index with a[i] > target
    int lo = 0, hi = a.size();
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] <= target) lo = mid + 1; else hi = mid;
    }
    return lo;
}
pair<int,int> firstLast(vector<int>& a, int x) {
    int lb = lowerBound(a, x), ub = upperBound(a, x);
    if (lb == (int)a.size() || a[lb] != x) return {-1, -1};
    return {lb, ub - 1};                          // count = ub - lb
}

Search in Rotated Sorted Array II (duplicates)

bool searchRotatedII(vector<int>& a, int target) {
    int lo = 0, hi = a.size() - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] == target) return true;
        if (a[lo] == a[mid] && a[mid] == a[hi]) { lo++; hi--; continue; }
        if (a[lo] <= a[mid]) {
            if (a[lo] <= target && target < a[mid]) hi = mid - 1; else lo = mid + 1;
        } else {
            if (a[mid] < target && target <= a[hi]) lo = mid + 1; else hi = mid - 1;
        }
    }
    return false;
}

Single Element in Sorted Array (others twice), index parity

int singleNonDuplicate(vector<int>& a) {
    int lo = 0, hi = a.size() - 1;
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (mid % 2) mid--;                     // keep mid even
        if (a[mid] == a[mid + 1]) lo = mid + 2; else hi = mid;
    }
    return a[lo];
}

Sqrt(x)

int mySqrt(int x) {
    long long lo = 0, hi = x, ans = 0;
    while (lo <= hi) {
        long long mid = lo + (hi - lo) / 2;
        if (mid * mid <= x) { ans = mid; lo = mid + 1; } else hi = mid - 1;
    }
    return ans;
}

Koko Eating Bananas, minimize speed

int minEatingSpeed(vector<int>& piles, int h) {
    int lo = 1, hi = *max_element(piles.begin(), piles.end());
    auto hours = [&](int speed) { long long t = 0; for (int p : piles) t += (p + speed - 1) / speed; return t; };
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (hours(mid) <= h) hi = mid; else lo = mid + 1;
    }
    return lo;
}

Ship Packages in D Days (also Allocate Books / Painter's Partition, same code: w = pages, days = students)

int shipWithinDays(vector<int>& w, int days) {
    int lo = *max_element(w.begin(), w.end()), hi = accumulate(w.begin(), w.end(), 0);
    auto daysNeeded = [&](int cap) {
        int d = 1, load = 0;
        for (int x : w) { if (load + x > cap) { d++; load = 0; } load += x; }
        return d;
    };
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (daysNeeded(mid) <= days) hi = mid; else lo = mid + 1;
    }
    return lo;
}

Aggressive Cows, maximize the minimum distance

int aggressiveCows(vector<int>& stalls, int cows) {
    sort(stalls.begin(), stalls.end());
    int lo = 1, hi = stalls.back() - stalls[0], ans = 0;
    auto can = [&](int dist) {
        int placed = 1, last = stalls[0];
        for (int i = 1; i < (int)stalls.size(); i++)
            if (stalls[i] - last >= dist) { placed++; last = stalls[i]; }
        return placed >= cows;
    };
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2;
        if (can(mid)) { ans = mid; lo = mid + 1; } else hi = mid - 1;
    }
    return ans;
}

Minimum Days to Make M Bouquets

int minDays(vector<int>& bloom, int m, int k) {
    if ((long long)m * k > (long long)bloom.size()) return -1;
    int lo = *min_element(bloom.begin(), bloom.end()), hi = *max_element(bloom.begin(), bloom.end());
    auto can = [&](int day) {
        int bouquets = 0, run = 0;
        for (int b : bloom) {
            if (b <= day) { if (++run == k) { bouquets++; run = 0; } }
            else run = 0;
        }
        return bouquets >= m;
    };
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (can(mid)) hi = mid; else lo = mid + 1;
    }
    return lo;
}

Smallest Divisor given a Threshold

int smallestDivisor(vector<int>& a, int threshold) {
    int lo = 1, hi = *max_element(a.begin(), a.end());
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        long long sum = 0;
        for (int x : a) sum += (x + mid - 1) / mid;
        if (sum <= threshold) hi = mid; else lo = mid + 1;
    }
    return lo;
}

Kth Missing Positive Number

int findKthPositive(vector<int>& a, int k) {
    int lo = 0, hi = a.size();
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (a[mid] - (mid + 1) < k) lo = mid + 1;   // missing numbers before a[mid]
        else hi = mid;
    }
    return lo + k;
}

Minimize Max Distance to Gas Station (real-valued BS)

double minMaxGasDist(vector<int>& st, int K) {
    double lo = 0, hi = 0;
    for (int i = 1; i < (int)st.size(); i++) hi = max(hi, (double)(st[i] - st[i - 1]));
    while (hi - lo > 1e-6) {
        double mid = (lo + hi) / 2;
        int need = 0;
        for (int i = 1; i < (int)st.size(); i++) need += (int)((st[i] - st[i - 1]) / mid);
        if (need <= K) hi = mid; else lo = mid;
    }
    return hi;
}

Matrix sorted like a flattened array (row = mid / C, col = mid % C)

bool searchMatrix1D(vector<vector<int>>& m, int target) {
    int R = m.size(), C = m[0].size(), lo = 0, hi = R * C - 1;
    while (lo <= hi) {
        int mid = lo + (hi - lo) / 2, val = m[mid / C][mid % C];
        if (val == target) return true;
        if (val < target) lo = mid + 1; else hi = mid - 1;
    }
    return false;
}

Row with Maximum 1's (rows sorted)

int rowWithMax1s(vector<vector<int>>& m) {
    int best = -1, maxOnes = 0;
    for (int r = 0; r < (int)m.size(); r++) {
        int ones = m[r].end() - lower_bound(m[r].begin(), m[r].end(), 1);
        if (ones > maxOnes) { maxOnes = ones; best = r; }
    }
    return best;
}

Kth Smallest in a Sorted Matrix, BS on value

int kthSmallestMatrix(vector<vector<int>>& m, int k) {
    int n = m.size(), lo = m[0][0], hi = m[n - 1][n - 1];
    auto countLE = [&](int x) {
        int cnt = 0, col = n - 1;
        for (int r = 0; r < n; r++) {
            while (col >= 0 && m[r][col] > x) col--;
            cnt += col + 1;
        }
        return cnt;
    };
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (countLE(mid) >= k) hi = mid; else lo = mid + 1;
    }
    return lo;
}

Matrix Median (odd R*C, rows sorted)

int matrixMedian(vector<vector<int>>& m) {
    int R = m.size(), C = m[0].size(), lo = INT_MAX, hi = INT_MIN;
    for (auto& row : m) { lo = min(lo, row[0]); hi = max(hi, row[C - 1]); }
    int need = (R * C + 1) / 2;
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2, cnt = 0;
        for (auto& row : m) cnt += upper_bound(row.begin(), row.end(), mid) - row.begin();
        if (cnt < need) lo = mid + 1; else hi = mid;
    }
    return lo;
}

Peak Element in 2D (values ≥ 0)

vector<int> findPeak2D(vector<vector<int>>& m) {
    int R = m.size(), C = m[0].size(), lo = 0, hi = C - 1;
    while (lo <= hi) {
        int mc = lo + (hi - lo) / 2, mr = 0;
        for (int r = 1; r < R; r++) if (m[r][mc] > m[mr][mc]) mr = r;
        int left  = mc > 0     ? m[mr][mc - 1] : -1;
        int right = mc < C - 1 ? m[mr][mc + 1] : -1;
        if (m[mr][mc] > left && m[mr][mc] > right) return {mr, mc};
        if (left > m[mr][mc]) hi = mc - 1; else lo = mc + 1;
    }
    return {-1, -1};
}

Median of Two Sorted Arrays, partition BS

double findMedianSortedArrays(vector<int>& A, vector<int>& B) {
    if (A.size() > B.size()) return findMedianSortedArrays(B, A);
    int n = A.size(), m = B.size(), lo = 0, hi = n, half = (n + m + 1) / 2;
    while (lo <= hi) {
        int i = lo + (hi - lo) / 2, j = half - i;
        int l1 = i > 0 ? A[i - 1] : INT_MIN, r1 = i < n ? A[i] : INT_MAX;
        int l2 = j > 0 ? B[j - 1] : INT_MIN, r2 = j < m ? B[j] : INT_MAX;
        if (l1 <= r2 && l2 <= r1) {
            if ((n + m) % 2) return max(l1, l2);
            return (max(l1, l2) + min(r1, r2)) / 2.0;
        }
        if (l1 > r2) hi = i - 1; else lo = i + 1;
    }
    return 0.0;
}

Kth Element of Two Sorted Arrays

int kthOfTwo(vector<int>& A, vector<int>& B, int k) {
    if (A.size() > B.size()) return kthOfTwo(B, A, k);
    int n = A.size(), m = B.size(), lo = max(0, k - m), hi = min(k, n);
    while (lo <= hi) {
        int i = lo + (hi - lo) / 2, j = k - i;
        int l1 = i > 0 ? A[i - 1] : INT_MIN, r1 = i < n ? A[i] : INT_MAX;
        int l2 = j > 0 ? B[j - 1] : INT_MIN, r2 = j < m ? B[j] : INT_MAX;
        if (l1 <= r2 && l2 <= r1) return max(l1, l2);
        if (l1 > r2) hi = i - 1; else lo = i + 1;
    }
    return -1;
}
