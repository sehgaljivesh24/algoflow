Recognition

When you see… Pattern
Single number, all others appear twice XOR every element (a ^ a = 0)
Two numbers appear once, rest appear twice XOR all → find a differing bit → split into two groups
One number appears three times, rest twice Bit counting mod 3
Count set bits from 0 to n DP: bits[i] = bits[i>>1] + (i & 1)
Generate all subsets / power set Bitmask, iterate 0 to 2ⁿ − 1
Minimum bit flips to convert A to B Count set bits in A ^ B
Maximum XOR of two numbers in an array Binary Trie over bits

Bit tricks cheat sheet

bool getBit(int n, int i)   { return (n >> i) & 1; }
int  setBit(int n, int i)   { return n | (1 << i); }
int  clearBit(int n, int i) { return n & ~(1 << i); }

Single Number — XOR trick

int singleNumber(vector<int>& a) {
    int result = 0;
    for (int x : a) result ^= x;
    return result;
}

Count Set Bits — Brian Kernighan's algorithm

int countSetBits(int n) {
    int count = 0;
    while (n) {
        n &= (n - 1);
        count++;
    }
    return count;
}

Single Number III (two unique numbers)

vector<int> singleNumberIII(vector<int>& a) {
    long long xr = 0;
    for (int x : a) xr ^= x;
    long long diff = xr & -xr;                   // a bit where the two answers differ
    int p = 0, q = 0;
    for (int x : a) { if (x & diff) p ^= x; else q ^= x; }
    return {p, q};
}

Single Number II (others appear 3 times), bit counting mod 3

int singleNumberII(vector<int>& a) {
    unsigned ans = 0;
    for (int bit = 0; bit < 32; bit++) {
        int cnt = 0;
        for (int x : a) cnt += (x >> bit) & 1;
        if (cnt % 3) ans |= (1u << bit);
    }
    return (int)ans;
}

Count Bits from 0 to n (DP)

vector<int> countBits(int n) {
    vector<int> bits(n + 1, 0);
    for (int i = 1; i <= n; i++) bits[i] = bits[i >> 1] + (i & 1);
    return bits;
}

Power Set via Bitmask

vector<vector<int>> powerSet(vector<int>& a) {
    int n = a.size();
    vector<vector<int>> res;
    for (int mask = 0; mask < (1 << n); mask++) {
        vector<int> subset;
        for (int i = 0; i < n; i++) if (mask & (1 << i)) subset.push_back(a[i]);
        res.push_back(subset);
    }
    return res;
}

Minimum Bit Flips A → B

int minBitFlips(int a, int b) { return __builtin_popcount(a ^ b); }

Divide Two Integers without * or /

int divideInts(int dividend, int divisor) {
    if (dividend == INT_MIN && divisor == -1) return INT_MAX;
    long long a = llabs((long long)dividend), b = llabs((long long)divisor), q = 0;
    while (a >= b) {
        long long t = b, shift = 1;
        while (a >= (t << 1)) { t <<= 1; shift <<= 1; }
        a -= t; q += shift;
    }
    return ((dividend < 0) ^ (divisor < 0)) ? -q : q;
}
