Recursion — Pattern-wise

Recognition

When you see… Pattern
Generate all subsets / subsequences Include/exclude recursion (pick / not-pick)
Subsets with duplicates, unique only Sort first, skip adjacent duplicates at the same recursion level
Combination Sum (reuse allowed) Pick/not-pick, stay at the same index when picking
Combination Sum II (no reuse, has duplicates) Sort, skip i > start && a[i]==a[i-1]
Permutations Swap-based recursion, or a visited[] array with backtracking
Permutations with duplicates Sort + visited[] + skip same-adjacent unless the previous was used
Palindrome partitioning Backtrack: try every prefix, recurse on the rest if it's a palindrome
N-Queens / Sudoku solver Backtracking with a placement-validity check
Word search in a grid DFS backtracking with visited marking/unmarking
Kth permutation sequence Factorial number system — no need to generate all

Example algorithms

Subsets — pick / not-pick

void solve(vector<int>& nums, int i, vector<int>& cur, vector<vector<int>>& res) {
    if (i == (int)nums.size()) { res.push_back(cur); return; }
    solve(nums, i + 1, cur, res);
    cur.push_back(nums[i]);
    solve(nums, i + 1, cur, res);
    cur.pop_back();
}

vector<vector<int>> subsets(vector<int>& nums) {
    vector<vector<int>> res;
    vector<int> cur;
    solve(nums, 0, cur, res);
    return res;
}

N-Queens — backtracking

bool isSafe(vector<string>& board, int row, int col, int n) {
    for (int i = 0; i < row; i++)
        if (board[i][col] == 'Q') return false;
    for (int i = row, j = col; i >= 0 && j >= 0; i--, j--)
        if (board[i][j] == 'Q') return false;
    for (int i = row, j = col; i >= 0 && j < n; i--, j++)
        if (board[i][j] == 'Q') return false;
    return true;
}

void solve(int row, int n, vector<string>& board, vector<vector<string>>& res) {
    if (row == n) { res.push_back(board); return; }
    for (int col = 0; col < n; col++) {
        if (isSafe(board, row, col, n)) {
            board[row][col] = 'Q';
            solve(row + 1, n, board, res);
            board[row][col] = '.';
        }
    }
}

Subsets II (duplicates)

void genSubsets(vector<int>& a, int start, vector<int>& cur, vector<vector<int>>& res) {
    res.push_back(cur);
    for (int i = start; i < (int)a.size(); i++) {
        if (i > start && a[i] == a[i - 1]) continue;
        cur.push_back(a[i]);
        genSubsets(a, i + 1, cur, res);
        cur.pop_back();
    }
}
vector<vector<int>> subsetsWithDup(vector<int>& a) {
    sort(a.begin(), a.end());
    vector<vector<int>> res; vector<int> cur;
    genSubsets(a, 0, cur, res);
    return res;
}

Combination Sum (reuse allowed)

void combo(vector<int>& a, int i, int target, vector<int>& cur, vector<vector<int>>& res) {
    if (target == 0) { res.push_back(cur); return; }
    if (i == (int)a.size() || target < 0) return;
    cur.push_back(a[i]);
    combo(a, i, target - a[i], cur, res);      // pick, stay at i
    cur.pop_back();
    combo(a, i + 1, target, cur, res);         // not pick
}

Combination Sum II (no reuse, duplicates), sort first

void combo2(vector<int>& a, int start, int target, vector<int>& cur, vector<vector<int>>& res) {
    if (target == 0) { res.push_back(cur); return; }
    for (int i = start; i < (int)a.size(); i++) {
        if (i > start && a[i] == a[i - 1]) continue;
        if (a[i] > target) break;
        cur.push_back(a[i]);
        combo2(a, i + 1, target - a[i], cur, res);
        cur.pop_back();
    }
}

Permutations, visited[] (II version: sort + the extra skip line)

void permute(vector<int>& a, vector<bool>& used, vector<int>& cur, vector<vector<int>>& res) {
    if (cur.size() == a.size()) { res.push_back(cur); return; }
    for (int i = 0; i < (int)a.size(); i++) {
        if (used[i]) continue;
        // Permutations II (a sorted): if (i > 0 && a[i] == a[i-1] && !used[i-1]) continue;
        used[i] = true; cur.push_back(a[i]);
        permute(a, used, cur, res);
        cur.pop_back(); used[i] = false;
    }
}

Palindrome Partitioning

bool isPal(const string& s, int l, int r) { while (l < r) if (s[l++] != s[r--]) return false; return true; }

void palPartition(const string& s, int start, vector<string>& cur, vector<vector<string>>& res) {
    if (start == (int)s.size()) { res.push_back(cur); return; }
    for (int end = start; end < (int)s.size(); end++) {
        if (!isPal(s, start, end)) continue;
        cur.push_back(s.substr(start, end - start + 1));
        palPartition(s, end + 1, cur, res);
        cur.pop_back();
    }
}

Word Search in a Grid

bool wordDfs(vector<vector<char>>& b, const string& w, int i, int j, int k) {
    if (k == (int)w.size()) return true;
    if (i < 0 || j < 0 || i >= (int)b.size() || j >= (int)b[0].size() || b[i][j] != w[k]) return false;
    char tmp = b[i][j];
    b[i][j] = '#';
    bool found = wordDfs(b, w, i + 1, j, k + 1) || wordDfs(b, w, i - 1, j, k + 1)
              || wordDfs(b, w, i, j + 1, k + 1) || wordDfs(b, w, i, j - 1, k + 1);
    b[i][j] = tmp;
    return found;
}
bool exist(vector<vector<char>>& b, string w) {
    for (int i = 0; i < (int)b.size(); i++)
        for (int j = 0; j < (int)b[0].size(); j++)
            if (wordDfs(b, w, i, j, 0)) return true;
    return false;
}

Sudoku Solver

bool canPlace(vector<vector<char>>& b, int r, int c, char ch) {
    for (int i = 0; i < 9; i++) {
        if (b[r][i] == ch || b[i][c] == ch) return false;
        if (b[3 * (r / 3) + i / 3][3 * (c / 3) + i % 3] == ch) return false;
    }
    return true;
}
bool solveSudoku(vector<vector<char>>& b) {
    for (int r = 0; r < 9; r++)
        for (int c = 0; c < 9; c++) {
            if (b[r][c] != '.') continue;
            for (char ch = '1'; ch <= '9'; ch++) {
                if (!canPlace(b, r, c, ch)) continue;
                b[r][c] = ch;
                if (solveSudoku(b)) return true;
                b[r][c] = '.';
            }
            return false;
        }
    return true;
}

Kth Permutation Sequence (factorial number system)

string getPermutation(int n, int k) {
    vector<int> fact(n, 1);
    for (int i = 1; i < n; i++) fact[i] = fact[i - 1] * i;
    string digits, res;
    for (int i = 1; i <= n; i++) digits += char('0' + i);
    k--;
    for (int i = n - 1; i >= 0; i--) {
        int idx = k / fact[i];
        res += digits[idx];
        digits.erase(digits.begin() + idx);
        k %= fact[i];
    }
    return res;
}
