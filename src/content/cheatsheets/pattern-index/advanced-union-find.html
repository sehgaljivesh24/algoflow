Example algorithms

Kruskal's Algorithm — MST

struct Edge {
    int u, v, weight;
    bool operator<(const Edge& other) const {
        return weight < other.weight;
    }
};

struct DSU {
    vector<int> parent, rank_;
    DSU(int n) : parent(n), rank_(n, 0) {
        iota(parent.begin(), parent.end(), 0);
    }
    int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }
    bool unite(int x, int y) {
        int rx = find(x), ry = find(y);
        if (rx == ry) return false;
        if (rank_[rx] < rank_[ry]) swap(rx, ry);
        parent[ry] = rx;
        if (rank_[rx] == rank_[ry]) rank_[rx]++;
        return true;
    }
};

long long kruskal(int n, vector<Edge>& edges) {
    sort(edges.begin(), edges.end());
    DSU dsu(n);
    long long mst_weight = 0;

    for (auto& e : edges) {
        if (dsu.unite(e.u, e.v)) {
            mst_weight += e.weight;
        }
    }
    return mst_weight;
}

Bipartite Checking with Union–Find

The classic trick: union each node with the opposite group of its neighbor. If an edge's two endpoints ever end up in the same group, that's a contradiction — not bipartite.

struct BipartiteDSU {
    vector<int> parent;
    int n;

    BipartiteDSU(int n_) : n(n_), parent(2 * n_) {
        iota(parent.begin(), parent.end(), 0);
    }

    int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]);
        return parent[x];
    }

    void unite(int x, int y) {
        parent[find(x)] = find(y);
    }

    // call once per edge (u, v): they must end up different colors.
    // returns false the moment bipartiteness is violated
    bool addEdge(int u, int v) {
        if (find(u) == find(v)) return false;  // already forced same color
        unite(u, v + n);   // u's color == v's opposite
        unite(v, u + n);   // v's color == u's opposite
        return true;
    }
};

Weighted Union–Find — Potential Graph

struct WeightedDSU {
    vector<int> parent;
    vector<long long> weight;

    WeightedDSU(int n) : parent(n), weight(n, 0) {
        iota(parent.begin(), parent.end(), 0);
    }

    pair<int, long long> find(int x) {
        if (parent[x] == x) return {x, 0};
        auto [root, w] = find(parent[x]);
        weight[x] += w;
        parent[x] = root;
        return {root, weight[x]};
    }

    bool unite(int x, int y, long long w) {
        auto [rx, wx] = find(x);
        auto [ry, wy] = find(y);
        if (rx == ry) return false;

        parent[ry] = rx;
        weight[ry] = wx - wy + w;
        return true;
    }
};

Cycle Detection / Redundant Connection (uses the DSU with bool unite from Kruskal)

vector<int> findRedundantConnection(vector<vector<int>>& edges) {
    DSU dsu(edges.size() + 1);
    for (auto& e : edges)
        if (!dsu.unite(e[0], e[1])) return e;        // already connected => this edge closes a cycle
    return {};
}

DSU with Component Size and Component Count

struct DSUSize {
    vector<int> parent, sz; int components;
    DSUSize(int n) : parent(n), sz(n, 1), components(n) { iota(parent.begin(), parent.end(), 0); }
    int find(int x) { return parent[x] == x ? x : parent[x] = find(parent[x]); }
    bool unite(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b);
        parent[b] = a; sz[a] += sz[b]; components--;
        return true;
    }
    int size(int x) { return sz[find(x)]; }
};
