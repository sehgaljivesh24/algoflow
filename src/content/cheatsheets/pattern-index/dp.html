Dynamic Programming

Overlapping subproblems + optimal substructure. Ask “what changes as I move forward?” — that becomes your state.

Pattern recognition

When you see… Pattern
Ways to reach step N — depends linearly on i−1, i−2 1D DP (climbing stairs, house robber)
Paths in a grid, min/max path sum 2D grid DP
Include or exclude an item, weight limit 0/1 Knapsack (subset sum, equal partition)
Unlimited use of items Unbounded Knapsack (coin change, rod cutting)
Two strings, common subsequence/substring LCS family (LCS, LPS, edit distance)
Longest increasing subsequence LIS — O(n²) or O(n log n) with binary search
Count/find a subsequence satisfying a condition DP on subsequences (distinct subsequences)
Partition or count palindromic substrings DP on strings / palindromes
Optimal way to split or parenthesize a range Interval DP (matrix chain multiplication, burst balloons)
Max path sum or diameter in a tree DP on trees — see Advanced DP
Assign N items to N slots optimally, small N (≤20) Bitmask DP (TSP, assignment problem) — see Advanced DP
Count numbers in a range with a digit property Digit DP — see Advanced DP
Buy/sell stock with cooldown, fee, or K transactions State machine DP

Example algorithms

0/1 Knapsack

int knapsack(vector<int>& wt, vector<int>& val, int W) {
    int n = wt.size();
    vector<vector<int>> dp(n + 1, vector<int>(W + 1, 0));
    for (int i = 1; i <= n; i++) {
        for (int w = 0; w <= W; w++) {
            dp[i][w] = dp[i - 1][w];
            if (wt[i - 1] <= w)
                dp[i][w] = max(dp[i][w], val[i - 1] + dp[i - 1][w - wt[i - 1]]);
        }
    }
    return dp[n][W];
}

Longest Common Subsequence

int longestCommonSubsequence(string a, string b) {
    int n = a.size(), m = b.size();
    vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            dp[i][j] = (a[i - 1] == b[j - 1])
                ? 1 + dp[i - 1][j - 1]
                : max(dp[i - 1][j], dp[i][j - 1]);
    return dp[n][m];
}

Longest Increasing Subsequence — O(n log n)

int lengthOfLIS(vector<int>& a) {
    vector<int> tails;
    for (int x : a) {
        auto it = lower_bound(tails.begin(), tails.end(), x);
        if (it == tails.end()) tails.push_back(x);
        else *it = x;
    }
    return tails.size();
}

Coin Change — unbounded knapsack, min coins

int coinChange(vector<int>& coins, int amount) {
    vector<int> dp(amount + 1, INT_MAX - 1);
    dp[0] = 0;
    for (int a = 1; a <= amount; a++)
        for (int c : coins)
            if (c <= a) dp[a] = min(dp[a], 1 + dp[a - c]);
    return dp[amount] >= INT_MAX - 1 ? -1 : dp[amount];
}

Matrix Chain Multiplication — interval DP

The template for every “optimal way to split a range” problem: try every split point k, combine the cost of both halves plus the cost of merging them.

int matrixChainOrder(vector<int>& dims) {
    int n = dims.size() - 1; // n matrices, dims has n+1 entries
    vector<vector<int>> dp(n + 1, vector<int>(n + 1, 0));

    for (int len = 2; len <= n; len++) {
        for (int i = 1; i <= n - len + 1; i++) {
            int j = i + len - 1;
            dp[i][j] = INT_MAX;
            for (int k = i; k < j; k++) {
                int cost = dp[i][k] + dp[k + 1][j]
                         + dims[i - 1] * dims[k] * dims[j];
                dp[i][j] = min(dp[i][j], cost);
            }
        }
    }
    return dp[1][n];
}

Climbing Stairs and House Robber (1D, O(1) space)

int climbStairs(int n) {
    int a = 1, b = 1;
    for (int i = 2; i <= n; i++) { int c = a + b; a = b; b = c; }
    return b;
}
int houseRobber(vector<int>& a) {
    int take = 0, skip = 0;
    for (int x : a) { int newTake = skip + x; skip = max(skip, take); take = newTake; }
    return max(take, skip);
}

Grid DP: Unique Paths and Min Path Sum

int uniquePaths(int m, int n) {
    vector<int> dp(n, 1);
    for (int i = 1; i < m; i++)
        for (int j = 1; j < n; j++) dp[j] += dp[j - 1];
    return dp[n - 1];
}
int minPathSum(vector<vector<int>>& g) {
    int R = g.size(), C = g[0].size();
    vector<vector<int>> dp(R, vector<int>(C, 0));
    for (int i = 0; i < R; i++)
        for (int j = 0; j < C; j++) {
            if (!i && !j) dp[i][j] = g[i][j];
            else dp[i][j] = g[i][j] + min(i ? dp[i - 1][j] : INT_MAX, j ? dp[i][j - 1] : INT_MAX);
        }
    return dp[R - 1][C - 1];
}

Subset Sum / Partition Equal Subset Sum (0/1)

bool canPartition(vector<int>& a) {
    int sum = accumulate(a.begin(), a.end(), 0);
    if (sum % 2) return false;
    int target = sum / 2;
    vector<bool> dp(target + 1, false); dp[0] = true;
    for (int x : a)
        for (int s = target; s >= x; s--) dp[s] = dp[s] || dp[s - x];
    return dp[target];
}

Coin Change II (number of ways, unbounded)

long long coinWays(int amount, vector<int>& coins) {
    vector<long long> dp(amount + 1, 0); dp[0] = 1;
    for (int c : coins)                         // coins outer loop = combinations, not permutations
        for (int a = c; a <= amount; a++) dp[a] += dp[a - c];
    return dp[amount];
}

Edit Distance

int editDistance(string a, string b) {
    int n = a.size(), m = b.size();
    vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
    for (int i = 0; i <= n; i++) dp[i][0] = i;
    for (int j = 0; j <= m; j++) dp[0][j] = j;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            dp[i][j] = a[i - 1] == b[j - 1] ? dp[i - 1][j - 1]
                     : 1 + min({dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1]});
    return dp[n][m];
}

Longest Palindromic Subsequence and Count Palindromic Substrings

int longestPalindromeSubseq(string s) {
    string r(s.rbegin(), s.rend());
    return longestCommonSubsequence(s, r);        // LCS from above
}
int countSubstrings(string s) {
    int n = s.size(), count = 0;
    vector<vector<bool>> dp(n, vector<bool>(n, false));
    for (int i = n - 1; i >= 0; i--)
        for (int j = i; j < n; j++)
            if (s[i] == s[j] && (j - i < 2 || dp[i + 1][j - 1])) { dp[i][j] = true; count++; }
    return count;
}

Distinct Subsequences

long long numDistinct(string s, string t) {
    int n = s.size(), m = t.size();
    vector<vector<long long>> dp(n + 1, vector<long long>(m + 1, 0));
    for (int i = 0; i <= n; i++) dp[i][0] = 1;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++) {
            dp[i][j] = dp[i - 1][j];
            if (s[i - 1] == t[j - 1]) dp[i][j] += dp[i - 1][j - 1];
        }
    return dp[n][m];
}

Burst Balloons (interval DP, k = last balloon)

int maxCoins(vector<int>& nums) {
    vector<int> a = {1};
    a.insert(a.end(), nums.begin(), nums.end());
    a.push_back(1);
    int n = a.size();
    vector<vector<int>> dp(n, vector<int>(n, 0));
    for (int len = 2; len < n; len++)
        for (int i = 0; i + len < n; i++) {
            int j = i + len;
            for (int k = i + 1; k < j; k++)
                dp[i][j] = max(dp[i][j], dp[i][k] + a[i] * a[k] * a[j] + dp[k][j]);
        }
    return dp[0][n - 1];
}

Buy/Sell Stock with Cooldown (state machine)

int maxProfitCooldown(vector<int>& p) {
    int hold = INT_MIN, sold = 0, rest = 0;
    for (int price : p) {
        int prevSold = sold;
        sold = hold + price;
        hold = max(hold, rest - price);
        rest = max(rest, prevSold);
    }
    return max(sold, rest);
}
// With transaction fee: hold = max(hold, cash - price); cash = max(cash, hold + price - fee);

Brute → Memo → Tabulation: 7 core pillars

Write each pillar in three steps: first recursion (brute force), then memoization, then tabulation. The arguments that change in the recursion become the loop variables in the tabulation.

Pillar 1 - 0/1 Knapsack (Subset Sum)

Core idea: For each element you have two choices: Take it or Not Take it. Each item can be used only once. Standard problem: Subset Sum (is there a subset whose sum equals the target?).

1. Brute Force (Recursion)

// Call: solve(n - 1, k, arr)
bool solve(int ind, int target, vector<int>& arr) {
    // Base Cases
    if (target == 0) return true;          // Target found
    if (ind == 0) return arr[0] == target; // Last element: does it equal the target?

    // Choice 1: Not Take
    bool notTake = solve(ind - 1, target, arr);

    // Choice 2: Take (only if the element is <= target)
    bool take = false;
    if (arr[ind] <= target)
        take = solve(ind - 1, target - arr[ind], arr);

    return take || notTake;
}

2. Memoization (Top-Down 2D DP)

Two states change here: ind and target. So we need a 2D DP table.

// Call: vector<vector<int>> dp(n, vector<int>(k + 1, -1)); solve(n - 1, k, arr, dp);
bool solve(int ind, int target, vector<int>& arr, vector<vector<int>>& dp) {
    if (target == 0) return true;
    if (ind == 0) return arr[0] == target;

    if (dp[ind][target] != -1) return dp[ind][target];

    bool notTake = solve(ind - 1, target, arr, dp);
    bool take = false;
    if (arr[ind] <= target)
        take = solve(ind - 1, target - arr[ind], arr, dp);

    return dp[ind][target] = (take || notTake);
}

3. Tabulation (Bottom-Up 2D DP)

Conversion logic: make an n x (target + 1) table. Base cases: for target == 0, set every row to true; at ind == 0, set dp[0][arr[0]] = true. Loops: ind from 1 to n-1, target from 1 to k.

bool subsetSumToK(int n, int k, vector<int>& arr) {
    vector<vector<bool>> dp(n, vector<bool>(k + 1, false));

    // Base case 1: target 0 is always true
    for (int i = 0; i < n; i++) dp[i][0] = true;

    // Base case 2: at index 0, if the element is <= target
    if (arr[0] <= k) dp[0][arr[0]] = true;

    // Bottom-up loops
    for (int ind = 1; ind < n; ind++) {
        for (int target = 1; target <= k; target++) {
            bool notTake = dp[ind - 1][target];
            bool take = false;
            if (arr[ind] <= target)
                take = dp[ind - 1][target - arr[ind]];
            dp[ind][target] = take || notTake;
        }
    }
    return dp[n - 1][k];
}

Pillar 2 - Unbounded Knapsack (Coin Change II)

Core idea: Same as 0/1 Knapsack, but you can use an element any number of times. Standard problem: Coin Change II (count the ways to make the target amount).

1. Brute Force (Recursion)

// Call: solve(n - 1, amount, coins)
int solve(int ind, int T, vector<int>& coins) {
    if (ind == 0)
        return (T % coins[0] == 0); // if the coin divides the target exactly, that is 1 way

    int notTake = solve(ind - 1, T, coins);
    int take = 0;
    if (coins[ind] <= T) {
        // NOTE: we keep 'ind' (not ind - 1) because we can use the same coin again
        take = solve(ind, T - coins[ind], coins);
    }
    return take + notTake;
}

2. Memoization (Top-Down 2D DP)

// Call: vector<vector<int>> dp(n, vector<int>(amount + 1, -1)); solve(n - 1, amount, coins, dp);
int solve(int ind, int T, vector<int>& coins, vector<vector<int>>& dp) {
    if (ind == 0) return (T % coins[0] == 0);
    if (dp[ind][T] != -1) return dp[ind][T];

    int notTake = solve(ind - 1, T, coins, dp);
    int take = 0;
    if (coins[ind] <= T)
        take = solve(ind, T - coins[ind], coins, dp);

    return dp[ind][T] = take + notTake;
}

3. Tabulation (Bottom-Up 2D DP)

Conversion logic: turn the same recursion into two loops. ind goes from 1 to n-1, T goes from 0 to amount.

int change(int amount, vector<int>& coins) {
    int n = coins.size();
    vector<vector<int>> dp(n, vector<int>(amount + 1, 0));

    // Base case initialization
    for (int T = 0; T <= amount; T++)
        if (T % coins[0] == 0) dp[0][T] = 1;

    for (int ind = 1; ind < n; ind++) {
        for (int T = 0; T <= amount; T++) {
            int notTake = dp[ind - 1][T];
            int take = 0;
            if (coins[ind] <= T)
                take = dp[ind][T - coins[ind]]; // same row 'ind' is used (unbounded)
            dp[ind][T] = take + notTake;
        }
    }
    return dp[n - 1][amount];
}

Pillar 3 - LCS / String DP

Core idea: Move two pointers (i, j) over the two strings. If the characters match, add 1 and move both back; if not, try both options. Standard problem: Longest Common Subsequence.

1. Brute Force (Recursion)

// Call: solve(n - 1, m - 1, s1, s2)
int solve(int i, int j, string& s1, string& s2) {
    if (i < 0 || j < 0) return 0; // One string is finished

    // Match found
    if (s1[i] == s2[j])
        return 1 + solve(i - 1, j - 1, s1, s2);

    // No match: skip one character of s1, or one character of s2
    return max(solve(i - 1, j, s1, s2), solve(i, j - 1, s1, s2));
}

2. Memoization (Top-Down 2D DP)

Trick: to avoid negative indexes (i < 0), use index shifting: treat 0 like -1 and make the table size +1.

// Call: vector<vector<int>> dp(n + 1, vector<int>(m + 1, -1)); solve(n, m, s1, s2, dp);
int solve(int i, int j, string& s1, string& s2, vector<vector<int>>& dp) {
    if (i == 0 || j == 0) return 0; // Shifted base case
    if (dp[i][j] != -1) return dp[i][j];

    if (s1[i - 1] == s2[j - 1])
        return dp[i][j] = 1 + solve(i - 1, j - 1, s1, s2, dp);

    return dp[i][j] = max(solve(i - 1, j, s1, s2, dp), solve(i, j - 1, s1, s2, dp));
}

3. Tabulation (Bottom-Up 2D DP)

int longestCommonSubsequence(string s1, string s2) {
    int n = s1.size(), m = s2.size();
    vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));

    // Base cases are already 0 from the initialization

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            if (s1[i - 1] == s2[j - 1])
                dp[i][j] = 1 + dp[i - 1][j - 1];
            else
                dp[i][j] = max(dp[i - 1][j], dp[i][j - 1]);
        }
    }
    return dp[n][m];
}

Pillar 4 - Matrix Chain Multiplication / Partition DP

Core idea: Split an array or string at different points k and pick the best answer. The loops run between i and j. Standard problem: MCM (minimize the multiplication cost).

1. Brute Force (Recursion)

// Call: solve(1, N - 1, arr)
int solve(int i, int j, vector<int>& arr) {
    if (i == j) return 0; // A single matrix costs 0

    int mini = INT_MAX;
    // Try every partition point 'k' from i to j-1
    for (int k = i; k < j; k++) {
        int steps = (arr[i - 1] * arr[k] * arr[j]) + solve(i, k, arr) + solve(k + 1, j, arr);
        mini = min(mini, steps);
    }
    return mini;
}

2. Memoization (Top-Down 2D DP)

// Call: vector<vector<int>> dp(N, vector<int>(N, -1)); solve(1, N - 1, arr, dp);
int solve(int i, int j, vector<int>& arr, vector<vector<int>>& dp) {
    if (i == j) return 0;
    if (dp[i][j] != -1) return dp[i][j];

    int mini = INT_MAX;
    for (int k = i; k < j; k++) {
        int steps = (arr[i - 1] * arr[k] * arr[j]) + solve(i, k, arr, dp) + solve(k + 1, j, arr, dp);
        mini = min(mini, steps);
    }
    return dp[i][j] = mini;
}

3. Tabulation (Bottom-Up 2D DP)

Conversion logic: the state (i, j) depends on smaller ranges (i..k and k+1..j). So fill the table from small ranges to big ones: the i loop goes from N-1 down to 1, and the j loop goes from i+1 up to N-1.

int matrixMultiplication(vector<int>& arr, int N) {
    vector<vector<int>> dp(N, vector<int>(N, 0));

    // dp[i][i] is already 0
    for (int i = N - 1; i >= 1; i--) {
        for (int j = i + 1; j < N; j++) {
            int mini = INT_MAX;
            for (int k = i; k < j; k++) {
                int steps = (arr[i - 1] * arr[k] * arr[j]) + dp[i][k] + dp[k + 1][j];
                mini = min(mini, steps);
            }
            dp[i][j] = mini;
        }
    }
    return dp[1][N - 1];
}

Pillar 5 - DP on Stocks (State Machine)

Core idea: Each day you are in one of two states: you can Buy (buy == 1), or you can only Sell (buy == 0). Standard problem: Buy and Sell Stock II (unlimited transactions).

1. Brute Force (Recursion)

// Call: solve(0, 1, prices)
int solve(int ind, int buy, vector<int>& prices) {
    if (ind == (int)prices.size()) return 0; // No more days

    int profit = 0;
    if (buy) {
        // Choice 1: buy today, Choice 2: skip today
        profit = max(-prices[ind] + solve(ind + 1, 0, prices), 0 + solve(ind + 1, 1, prices));
    } else {
        // Choice 1: sell today, Choice 2: skip today
        profit = max(prices[ind] + solve(ind + 1, 1, prices), 0 + solve(ind + 1, 0, prices));
    }
    return profit;
}

2. Memoization (Top-Down 2D DP)

States: ind (0 to n) and buy (0 or 1). Table size: n x 2.

// Call: vector<vector<int>> dp(n, vector<int>(2, -1)); solve(0, 1, prices, dp);
int solve(int ind, int buy, vector<int>& prices, vector<vector<int>>& dp) {
    if (ind == (int)prices.size()) return 0;
    if (dp[ind][buy] != -1) return dp[ind][buy];

    int profit = 0;
    if (buy)
        profit = max(-prices[ind] + solve(ind + 1, 0, prices, dp), solve(ind + 1, 1, prices, dp));
    else
        profit = max(prices[ind] + solve(ind + 1, 1, prices, dp), solve(ind + 1, 0, prices, dp));

    return dp[ind][buy] = profit;
}

3. Tabulation (Bottom-Up 2D DP)

int maxProfit(vector<int>& prices) {
    int n = prices.size();
    vector<vector<int>> dp(n + 1, vector<int>(2, 0));

    // Base case: dp[n][0] = dp[n][1] = 0 (already 0)

    for (int ind = n - 1; ind >= 0; ind--) {
        for (int buy = 0; buy <= 1; buy++) {
            int profit = 0;
            if (buy)
                profit = max(-prices[ind] + dp[ind + 1][0], dp[ind + 1][1]);
            else
                profit = max(prices[ind] + dp[ind + 1][1], dp[ind + 1][0]);
            dp[ind][buy] = profit;
        }
    }
    return dp[0][1]; // Start from day 0 with buying allowed
}

Pillar 6 - DP on Trees (House Robber III)

Core idea: Each node asks its children for their answers, then combines them (sum or max) to make its own decision. There are no loops: it works purely as a DFS (post-order). Standard problem: House Robber III (you cannot rob two directly connected nodes).

1. Brute Force (Recursion)

int rob(TreeNode* root) {
    if (!root) return 0;

    // Choice 1: rob this node -> skip its children, go to the grandchildren
    int val = root->val;
    if (root->left)  val += rob(root->left->left)  + rob(root->left->right);
    if (root->right) val += rob(root->right->left) + rob(root->right->right);

    // Choice 2: don't rob this node -> we can rob the left and right children
    int notRob = rob(root->left) + rob(root->right);

    return max(val, notRob);
}

2. Memoization (Top-Down using Hash Map)

Tree nodes have no sequential index, so instead of an array we use the Node* as the key of a hash map.

unordered_map<TreeNode*, int> dp; // call dp.clear() between test cases

int rob(TreeNode* root) {
    if (!root) return 0;
    if (dp.count(root)) return dp[root];

    int val = root->val;
    if (root->left)  val += rob(root->left->left)  + rob(root->left->right);
    if (root->right) val += rob(root->right->left) + rob(root->right->right);

    int notRob = rob(root->left) + rob(root->right);

    return dp[root] = max(val, notRob);
}

3. Tabulation Equivalent (Pair Approach)

Writing iterative tabulation for trees is hard without extra maps, so interviewers expect the pair-return solution: each node returns {robbed_this_node, not_robbed_this_node}. Time O(N), space O(H) (recursion stack).

class Solution {
public:
    // pair.first  = max money if we ROB this node
    // pair.second = max money if we DO NOT rob this node
    pair<int, int> solve(TreeNode* root) {
        if (!root) return {0, 0};

        pair<int, int> leftSub  = solve(root->left);
        pair<int, int> rightSub = solve(root->right);

        // Robbed this node -> its children cannot be robbed
        int robNode = root->val + leftSub.second + rightSub.second;

        // Did not rob this node -> for each child, take the better of the two
        int notRobNode = max(leftSub.first, leftSub.second) + max(rightSub.first, rightSub.second);

        return {robNode, notRobNode};
    }

    int rob(TreeNode* root) {
        pair<int, int> result = solve(root);
        return max(result.first, result.second);
    }
};

Pillar 7 - Digit DP

Core idea: You are asked to find numbers in a huge range [L, R] (where R can be up to 1018) that follow some condition, for example "digit sum equals X" or "no digit 4". A normal loop over 1018 numbers gives TLE, so instead we build the number digit by digit, from left to right. Standard problem: count the numbers between 0 and R whose digit sum is equal to X.

State Meaning
idx Which digit position we are at (left to right).
tight A 0/1 flag. If tight == 1, we are still bound by R and cannot pick a digit above R[idx]. If tight == 0, we can pick any digit from 0 to 9. Example with R = "324": at idx 0 the limit is 3. Picking 3 keeps tight = 1 (the next limit is 2); picking 0, 1 or 2 makes tight = 0 (free, limit 9).
condition state Whatever the question asks about (here: current_sum, the sum of the digits placed so far).

1. Brute Force (Recursive Construction)

Convert R into a string so we can loop over its digits.

// Call: solve(0, 1, 0, X, R)   where R = to_string(number), e.g. R = "324"
long long solve(int idx, int tight, int current_sum, int X, string& R) {
    // Base case: all digits are placed
    if (idx == (int)R.length())
        return (current_sum == X); // sum equals X -> we found 1 valid number

    long long ans = 0;

    // tight == 1: we cannot go above R[idx]
    // tight == 0: we are free (limit = 9)
    int limit = (tight == 1) ? (R[idx] - '0') : 9;

    // Try every possible digit at this position
    for (int digit = 0; digit <= limit; digit++) {
        // Stay tight only if we were tight AND picked the limit digit
        int new_tight = (tight == 1 && digit == limit) ? 1 : 0;

        ans += solve(idx + 1, new_tight, current_sum + digit, X, R);
    }
    return ans;
}

2. Memoization (Top-Down 3D DP)

States: idx (up to 19), tight (2), current_sum (max 18 x 9 = 162). That is only about 20 x 2 x 170 states, with 10 digit choices each. For a range [L, R], answer = solve(R) - solve(L - 1) (just solve(R) if L = 0). Reset dp before each call, because the tight = 1 states depend on the number. Use long long: the count can reach 10^18.

long long dp[20][2][170]; // global array

// Call: memset(dp, -1, sizeof(dp)); solve(0, 1, 0, X, R);
long long solve(int idx, int tight, int current_sum, int X, string& R) {
    if (idx == (int)R.length()) return (current_sum == X);

    if (dp[idx][tight][current_sum] != -1) return dp[idx][tight][current_sum];

    long long ans = 0;
    int limit = (tight == 1) ? (R[idx] - '0') : 9;

    for (int digit = 0; digit <= limit; digit++) {
        int new_tight = (tight == 1 && digit == limit) ? 1 : 0;
        ans += solve(idx + 1, new_tight, current_sum + digit, X, R);
    }

    return dp[idx][tight][current_sum] = ans;
}

3. Tabulation (Bottom-Up 3D DP)

Tabulation is a little tricky in Digit DP. The base case is at idx == n, so the idx loop runs backwards, from n-1 down to 0.

// Call: countWays(to_string(number), X)
long long countWays(string R, int X) {
    int n = R.length();
    int maxSum = 9 * n;        // biggest possible digit sum
    if (X > maxSum) return 0;  // impossible, and keeps array indexes safe

    // dp[idx][tight][current_sum]
    vector<vector<vector<long long>>> dp(
        n + 1, vector<vector<long long>>(2, vector<long long>(maxSum + 1, 0)));

    // Base case (idx == n): all digits placed and sum == X -> 1 way
    for (int t = 0; t <= 1; t++)
        dp[n][t][X] = 1;

    // Bottom-up loops
    for (int idx = n - 1; idx >= 0; idx--) {
        for (int tight = 0; tight <= 1; tight++) {
            for (int current_sum = 0; current_sum <= maxSum; current_sum++) {

                int limit = (tight == 1) ? (R[idx] - '0') : 9;
                long long ans = 0;

                for (int digit = 0; digit <= limit; digit++) {
                    int new_tight = (tight == 1 && digit == limit) ? 1 : 0;

                    if (current_sum + digit <= X) // boundary check (also keeps the index <= maxSum)
                        ans += dp[idx + 1][new_tight][current_sum + digit];
                }
                dp[idx][tight][current_sum] = ans;
            }
        }
    }

    // The answer is always at idx = 0, tight = 1, current_sum = 0
    return dp[0][1][0];
}

Interview cheat-code for Digit DP

Whenever the constraint is R ≥ 109 and the task is "count the numbers that satisfy some property", a normal array DP will fail. Think Digit DP.

Golden summary: converting recursion to tabulation

Step Rule
Check base cases Put the recursion's terminal return values into the starting cells of the DP table.
Loop direction If the recursion goes from n down to 0, the tabulation loop goes from 0 up to n (and the other way round).
Variable mapping The arguments that change in the recursion (ind, target, buy) become the loop variables of the tabulation (i, j, k).
